-- 1. جدول المتاجر المسجلة (stores)
CREATE TABLE IF NOT EXISTS public.stores (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    store_name TEXT NOT NULL,
    owner_name TEXT NOT NULL,
    phone TEXT NOT NULL,
    email TEXT,
    business_type TEXT DEFAULT 'سوبرماركت وبقالة',
    address TEXT DEFAULT 'السودان',
    status TEXT DEFAULT 'trial', -- 'trial', 'monthly', 'annual', 'expired', 'blocked'
    trial_started_at TIMESTAMPTZ DEFAULT now(),
    subscription_expires_at TIMESTAMPTZ NOT NULL,
    is_blocked BOOLEAN DEFAULT false,
    created_at TIMESTAMPTZ DEFAULT now()
);

-- 2. جدول طلبات التحويل البنكي والاشتراك (subscription_requests)
CREATE TABLE IF NOT EXISTS public.subscription_requests (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    store_id UUID REFERENCES public.stores(id) ON DELETE CASCADE,
    plan_type TEXT NOT NULL, -- 'monthly' (50,000) or 'annual' (500,000)
    amount NUMERIC NOT NULL,
    transfer_account TEXT DEFAULT '59189',
    account_holder TEXT DEFAULT 'محمد ياسر إبراهيم محمد',
    last4_digits TEXT,
    receipt_image_url TEXT,
    payment_method TEXT DEFAULT 'بنكك - بنك الخرطوم',
    status TEXT DEFAULT 'pending', -- 'pending', 'approved', 'rejected'
    notes TEXT,
    created_at TIMESTAMPTZ DEFAULT now(),
    approved_at TIMESTAMPTZ
);

-- 3. تفعيل الحماية RLS
ALTER TABLE public.stores ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.subscription_requests ENABLE ROW LEVEL SECURITY;
